\textbf{$R(\alpha,u_{x},u_{y},u_{z})$}\\[6pt]
\[
R(\alpha, u_{x}, u_{y}, u_{z}) =
\begin{pmatrix}
\frac{u_{x}^2+\cos (\alpha ) \left(u_{y}^2+u_{z}^2\right)}{u_{\mathrm{mag}}^2} & \frac{-u_{\mathrm{mag}} u_{z} \sin (\alpha )-u_{x} u_{y} \cos (\alpha )+u_{x} u_{y}}{u_{\mathrm{mag}}^2} & \frac{u_{\mathrm{mag}} u_{y} \sin (\alpha )-u_{x} u_{z} \cos (\alpha )+u_{x} u_{z}}{u_{\mathrm{mag}}^2} \\
\frac{u_{\mathrm{mag}} u_{z} \sin (\alpha )-u_{x} u_{y} \cos (\alpha )+u_{x} u_{y}}{u_{\mathrm{mag}}^2} & \frac{\cos (\alpha ) \left(u_{x}^2+u_{z}^2\right)+u_{y}^2}{u_{\mathrm{mag}}^2} & \frac{-u_{\mathrm{mag}} u_{x} \sin (\alpha )-u_{y} u_{z} \cos (\alpha )+u_{y} u_{z}}{u_{\mathrm{mag}}^2} \\
\frac{-u_{\mathrm{mag}} u_{y} \sin (\alpha )-u_{x} u_{z} \cos (\alpha )+u_{x} u_{z}}{u_{\mathrm{mag}}^2} & \frac{u_{\mathrm{mag}} u_{x} \sin (\alpha )-u_{y} u_{z} \cos (\alpha )+u_{y} u_{z}}{u_{\mathrm{mag}}^2} & \frac{\cos (\alpha ) \left(u_{x}^2+u_{y}^2\right)+u_{z}^2}{u_{\mathrm{mag}}^2}
\end{pmatrix}
\]
where $u_{\mathrm{mag}} = \sqrt{u_x^{2}+u_y^{2}+u_z^{2}}$.
\end{center}
